# Pre-release Log Analyzer — Allowlist of known false positives
#
# Format: category:pattern=severity
# Lines starting with # are comments, blank lines are ignored.
# Patterns are shell globs matched against the message field of findings.
# Severity overrides: ignore (drop finding), warning, info
# If severity is omitted, defaults to "ignore".
#
# The category field must match one of the analyzer categories:
#   crash, deprecation, version, slow_startup, connection,
#   security, memory, config, dependency

# ZK non-TLS quorum communication is expected in CI environments
security:*insecure*quorum*=ignore
security:*non-tls*quorum*=ignore

# JVM startup flags like -XX:+CrashOnOutOfMemoryError are not actual OOM errors
memory:*CrashOnOutOfMemoryError*=ignore

# Volume permission errors are expected in CI (tmpfs volumes are root-owned,
# application containers run as non-root UIDs)
security:*PermissionError*=warning
security:*permission denied*read-only*=warning

# Thumbnail directory write errors in CI — tmpfs thumbnail-data volume is root-owned
# but document-indexer runs as UID 1000. Thumbnails are non-critical; indexing succeeds.
security:*permission denied*/data/thumbnails/*=ignore

# Solr/JVM startup deprecations from the Solr 10 image and bundled Java stack.
# These are upstream runtime notices, not Aithena configuration knobs.
deprecation:*terminally deprecated method in sun.misc.Unsafe*=info
deprecation:*sun.misc.Unsafe::arrayBaseOffset will be removed in a future release*=info

# RabbitMQ 4.0-management emits a known startup deprecation notice for the
# management_metrics_collection feature even when the project does not set the
# deprecated_features.permit.management_metrics_collection flag. Keep the
# exact notice visible as info; unrelated RabbitMQ deprecations remain warnings.
deprecation:*management_metrics_collection*=info
deprecation:*will not be permitted by default in a future minor RabbitMQ version*future major RabbitMQ version*=info

# Connection retries during startup are normal for Docker Compose orchestration
connection:*retrying*=ignore
connection:*connection refused*=ignore

# ZooKeeper client/secure/observer ports are intentionally supplied through
# ZOO_SERVERS in the Compose ensemble, not separate static config keys.
config:*clientPort is not set*=ignore
config:*secureClientPort is not set*=ignore
config:*observerMasterPort is not set*=ignore

# ZooKeeper maxCnxns defaults to unlimited when not explicitly configured.
# The Compose services set ZOO_MAX_CLIENT_CNXNS=60, so this startup line is
# informational noise from the image bootstrap path.
config:*maxCnxns is not configured*=ignore

# Solr's default ZooKeeper credentials/ACL providers leave znodes open inside
# the private Compose network. This is an accepted dev/CI and default production
# posture until Brett implements optional ZK ACL hardening; do not hide any
# other authentication or authorization failures.
config:*Using default ZkCredentialsInjector*OPEN_ACL_UNSAFE*=ignore
config:*Using default ZkCredentialsProvider*=ignore
config:*Using default ZkACLProvider*=ignore
